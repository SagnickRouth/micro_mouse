cmake_minimum_required(VERSION 3.20)
project(micro_mouse VERSION 0.2.0 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_SYSTEM_NAME Generic)
set(CMAKE_SYSTEM_PROCESSOR arm)
set(CMAKE_C_COMPILER arm-none-eabi-gcc)
set(CMAKE_ASM_COMPILER arm-none-eabi-gcc)
set(CMAKE_OBJCOPY arm-none-eabi-objcopy)
set(CMAKE_SIZE arm-none-eabi-size)
set(MCU_FLAGS "-mcpu=cortex-m4 -mthumb -mfpu=fpv4-sp-d16 -mfloat-abi=hard")
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} ${MCU_FLAGS} -Wall -Wextra -Os -ffunction-sections -fdata-sections")
set(CMAKE_ASM_FLAGS "${CMAKE_ASM_FLAGS} ${MCU_FLAGS}")
file(GLOB SOURCES "src/*.c")
include_directories(include)
add_executable(${PROJECT_NAME}.elf ${SOURCES})
# Supply STM32CubeF4 HAL/CMSIS startup + linker files and the ST VL53L0X API
# through your CubeIDE/CubeMX project target sources/includes.
set_target_properties(${PROJECT_NAME}.elf PROPERTIES OUTPUT_NAME "micro_mouse")
